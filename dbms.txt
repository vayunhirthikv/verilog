/* =========================================================
   EXPERIMENT 5: SQL JOIN QUERIES
   ========================================================= */

CREATE TABLE DEPARTMENT (
    DEPT_ID INT PRIMARY KEY,
    DEPT_NAME VARCHAR2(50)
);

CREATE TABLE EMPLOYEE (
    EMP_ID INT PRIMARY KEY,
    EMP_NAME VARCHAR2(50),
    SALARY NUMBER(10,2),
    DEPT_ID INT,
    MANAGER_ID INT,
    FOREIGN KEY (DEPT_ID) REFERENCES DEPARTMENT(DEPT_ID)
);

INSERT INTO DEPARTMENT VALUES (101, 'HR');
INSERT INTO DEPARTMENT VALUES (102, 'Finance');
INSERT INTO DEPARTMENT VALUES (103, 'IT');
INSERT INTO DEPARTMENT VALUES (104, 'Marketing');

INSERT INTO EMPLOYEE VALUES (1, 'Amit', 50000, 101, NULL);
INSERT INTO EMPLOYEE VALUES (2, 'Priya', 60000, 102, 1);
INSERT INTO EMPLOYEE VALUES (3, 'Rahul', 55000, 103, 1);
INSERT INTO EMPLOYEE VALUES (4, 'Sneha', 45000, 103, 3);
INSERT INTO EMPLOYEE VALUES (5, 'Karan', 70000, NULL, 2);

-- INNER JOIN
SELECT E.EMP_ID, E.EMP_NAME, D.DEPT_NAME
FROM EMPLOYEE E
INNER JOIN DEPARTMENT D
ON E.DEPT_ID = D.DEPT_ID;

-- LEFT OUTER JOIN
SELECT E.EMP_NAME, D.DEPT_NAME
FROM EMPLOYEE E
LEFT OUTER JOIN DEPARTMENT D
ON E.DEPT_ID = D.DEPT_ID;

-- RIGHT OUTER JOIN
SELECT E.EMP_NAME, D.DEPT_NAME
FROM EMPLOYEE E
RIGHT OUTER JOIN DEPARTMENT D
ON E.DEPT_ID = D.DEPT_ID;

-- FULL OUTER JOIN
SELECT E.EMP_NAME, D.DEPT_NAME
FROM EMPLOYEE E
FULL OUTER JOIN DEPARTMENT D
ON E.DEPT_ID = D.DEPT_ID;

-- SELF JOIN
SELECT E.EMP_NAME AS EMPLOYEE,
       M.EMP_NAME AS MANAGER
FROM EMPLOYEE E
LEFT JOIN EMPLOYEE M
ON E.MANAGER_ID = M.EMP_ID;

-- JOIN WITH GROUP BY
SELECT D.DEPT_NAME, COUNT(E.EMP_ID) AS TOTAL_EMPLOYEES
FROM DEPARTMENT D
LEFT JOIN EMPLOYEE E
ON D.DEPT_ID = E.DEPT_ID
GROUP BY D.DEPT_NAME;

-- JOIN WITH SUBQUERY
SELECT E.EMP_NAME, D.DEPT_NAME, E.SALARY
FROM EMPLOYEE E
LEFT JOIN DEPARTMENT D
ON E.DEPT_ID = D.DEPT_ID
WHERE E.SALARY > (
    SELECT AVG(SALARY)
    FROM EMPLOYEE
);

CREATE TABLE AUDIT_EMPLOYEE (
    AUDIT_ID NUMBER PRIMARY KEY,
    EMP_ID NUMBER(5),
    EMP_NAME VARCHAR2(50),
    DEPARTMENT VARCHAR2(20),
    SALARY NUMBER(10,2),
    ACTION_TYPE VARCHAR2(20),
    ACTION_TIME DATE
);

CREATE SEQUENCE AUDIT_SEQ
START WITH 1
INCREMENT BY 1;

CREATE OR REPLACE TRIGGER TRG_SALARY_UPDATE
BEFORE UPDATE OF SALARY ON EMPLOYEE
FOR EACH ROW
BEGIN
    INSERT INTO AUDIT_EMPLOYEE
    VALUES (
        AUDIT_SEQ.NEXTVAL,
        :OLD.EMP_ID,
        :OLD.EMP_NAME,
        :OLD.DEPARTMENT,
        :OLD.SALARY,
        'SALARY UPDATE',
        SYSDATE
    );
END;
/

CREATE OR REPLACE TRIGGER TRG_EMPLOYEE_DELETE
BEFORE DELETE ON EMPLOYEE
FOR EACH ROW
BEGIN
    INSERT INTO AUDIT_EMPLOYEE
    VALUES (
        AUDIT_SEQ.NEXTVAL,
        :OLD.EMP_ID,
        :OLD.EMP_NAME,
        :OLD.DEPARTMENT,
        :OLD.SALARY,
        'DELETE',
        SYSDATE
    );
END;
/

-- Revenue generated by each product
SELECT P.PRODUCT_NAME,
       SUM(P.PRICE * S.QUANTITY) AS TOTAL_REVENUE
FROM PRODUCT P
JOIN SALES S ON P.PRODUCT_ID = S.PRODUCT_ID
GROUP BY P.PRODUCT_NAME;

-- Total revenue
SELECT SUM(P.PRICE * S.QUANTITY) AS TOTAL_REVENUE
FROM PRODUCT P
JOIN SALES S ON P.PRODUCT_ID = S.PRODUCT_ID;

-- Products generating revenue above average product revenue
SELECT PRODUCT_NAME, TOTAL_REVENUE
FROM (
    SELECT P.PRODUCT_NAME,
           SUM(P.PRICE * S.QUANTITY) AS TOTAL_REVENUE
    FROM PRODUCT P
    JOIN SALES S ON P.PRODUCT_ID = S.PRODUCT_ID
    GROUP BY P.PRODUCT_NAME
)
WHERE TOTAL_REVENUE > (
    SELECT AVG(TOTAL_REVENUE)
    FROM (
        SELECT SUM(P.PRICE * S.QUANTITY) AS TOTAL_REVENUE
        FROM PRODUCT P
        JOIN SALES S ON P.PRODUCT_ID = S.PRODUCT_ID
        GROUP BY P.PRODUCT_ID
    )
);

    DETAIL_ID NUMBER PRIMARY KEY,

-- Orders with total value greater than 50000
SELECT O.ORDER_ID,
       C.CUSTOMER_NAME,
       SUM(OD.QUANTITY * OD.PRICE) AS ORDER_TOTAL
FROM ORDERS O
JOIN CUSTOMER C ON O.CUSTOMER_ID = C.CUSTOMER_ID
JOIN ORDER_DETAILS OD ON O.ORDER_ID = OD.ORDER_ID
GROUP BY O.ORDER_ID, C.CUSTOMER_NAME
HAVING SUM(OD.QUANTITY * OD.PRICE) > 50000;

-- Orders with total value above average order value
SELECT ORDER_ID, ORDER_TOTAL
FROM (
    SELECT ORDER_ID, SUM(QUANTITY * PRICE) AS ORDER_TOTAL
    FROM ORDER_DETAILS
    GROUP BY ORDER_ID
)
WHERE ORDER_TOTAL > (
    SELECT AVG(ORDER_TOTAL)
    FROM (
        SELECT SUM(QUANTITY * PRICE) AS ORDER_TOTAL
        FROM ORDER_DETAILS
        GROUP BY ORDER_ID
    )
);

FROM CUSTOMER C
JOIN ORDERS O ON C.CUSTOMER_ID = O.CUSTOMER_ID
JOIN ORDER_ITEMS OI ON O.ORDER_ID = OI.ORDER_ID
JOIN PRODUCT P ON OI.PRODUCT_ID = P.PRODUCT_ID

    CASE
        WHEN OI.QUANTITY * P.PRICE > 2000 THEN O.ORDER_ID
    END
) >= 5;

-- Customers whose total spending exceeds average customer spending
SELECT CUSTOMER_NAME, TOTAL_SPENDING
FROM (
    SELECT C.CUSTOMER_ID, C.CUSTOMER_NAME,
           NVL(SUM(OI.QUANTITY * P.PRICE), 0) AS TOTAL_SPENDING
    FROM CUSTOMER C
    LEFT JOIN ORDERS O ON C.CUSTOMER_ID = O.CUSTOMER_ID
    LEFT JOIN ORDER_ITEMS OI ON O.ORDER_ID = OI.ORDER_ID
    LEFT JOIN PRODUCT P ON OI.PRODUCT_ID = P.PRODUCT_ID
    GROUP BY C.CUSTOMER_ID, C.CUSTOMER_NAME
)
WHERE TOTAL_SPENDING > (
    SELECT AVG(TOTAL_SPENDING)
    FROM (
        SELECT C.CUSTOMER_ID,
               NVL(SUM(OI.QUANTITY * P.PRICE), 0) AS TOTAL_SPENDING
        FROM CUSTOMER C
        LEFT JOIN ORDERS O ON C.CUSTOMER_ID = O.CUSTOMER_ID
        LEFT JOIN ORDER_ITEMS OI ON O.ORDER_ID = OI.ORDER_ID
        LEFT JOIN PRODUCT P ON OI.PRODUCT_ID = P.PRODUCT_ID
        GROUP BY C.CUSTOMER_ID
    )
);
