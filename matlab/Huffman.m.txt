clc;
clear;
close all;

% Input data
data = 'aaaabbbccc';
data = double(data);

% Find symbols and probabilities
symbols = unique(data);
N = length(data);
counts = zeros(1,length(symbols));

for i = 1:length(symbols)
    counts(i) = sum(data == symbols(i));
end

probs = counts/N;

% Huffman coding
dict = huffmandict(symbols,probs);

% Display dictionary
disp('Huffman Dictionary:');
for i = 1:length(symbols)
    fprintf('%c : ',symbols(i));
    disp(dict{i,2});
end

% Encode data
encoded = huffmanenco(data,dict);

disp('Encoded data:');
disp(encoded);

% Calculate results
H = -sum(probs .* log2(probs));
avgLen = sum(probs .* cellfun(@length,dict(:,2)));
compressedBits = length(encoded);
originalBits = N*8;

fprintf('Entropy = %.3f bits/symbol\n',H);
fprintf('Average code length = %.3f bits/symbol\n',avgLen);
fprintf('Original size = %d bits\n',originalBits);
fprintf('Compressed size = %d bits\n',compressedBits);
fprintf('Compression ratio = %.3f\n',originalBits/compressedBits);
