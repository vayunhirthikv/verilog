clc;
clear;
close all;

data = [1 0 1 1];
input = [data 0 0];

g1 = [1 1 1];
g2 = [1 0 1];

reg = [0 0];
code = [];
states = 1;     % Initial state 00

% Convolutional encoding
for i = 1:length(input)
    reg = [input(i) reg];

    o1 = mod(sum(reg .* g1), 2);
    o2 = mod(sum(reg .* g2), 2);
    code = [code o1 o2];

    reg = reg(1:2);

    % State: reg = [previous input, older input]
    states = [states, reg(1)*2 + reg(2) + 1];
end

disp('Input data:');
disp(data);
disp('Encoded data:');
disp(code);

% Trellis transitions
next = [1 3; 1 3; 2 4; 2 4];

figure;
hold on;

% Draw all possible paths
for t = 1:6
    for s = 1:4
        for b = 1:2
            ns = next(s,b);
            plot([t t+1], [s ns], 'k--');
        end
    end
end

% Mark actual path
plot(1:7, states, 'r-o', 'LineWidth', 3, ...
    'MarkerFaceColor', 'r', 'MarkerSize', 7);

yticks(1:4);
yticklabels({'00','01','10','11'});
xticks(1:7);
xlabel('Time step');
ylabel('State');
title('Convolutional Encoder Trellis with Actual Path');
grid on;
hold off;
