module fulladd(
input a,b,cin,
output sum,cout


    );
   assign sum=a^b^cin;
   assign cout=(a&b)|(cin&b)|(a&cin);
endmodule

module ripple(a,b,cin,sum,cout );
input [7:0]a,b;
input cin;
output [7:0] sum;
output cout;

wire c1,c2,c3,c4,c5,c6,c7;
fulladd f1(a[0],b[0],cin,sum[0],c1);
fulladd f2(a[1],b[1],c1,sum[1],c2);
fulladd f3(a[2],b[2],c2,sum[2],c3);
fulladd f4(a[3],b[3],c3,sum[3],c4);
fulladd f5(a[4],b[4],c4,sum[4],c5);
fulladd f6(a[5],b[5],c5,sum[5],c6);
fulladd f7(a[6],b[6],c6,sum[6],c7);
fulladd f8(a[7],b[7],c7,sum[7],cout);

endmodule

module ripple_tb;
reg [7:0]a,b;
reg cin;
wire [7:0] sum;
wire cout;
ripple uut(.a(a),.b(b),.cin(cin),.cout(cout),.sum(sum));
initial begin
$monitor("A=%d | B=%d | Cin=%b | Sum=%d |Cout=%b ",a,b,cin,sum,cout);
a=8'd1; b=8'd2; cin=1; #10;
a=8'd2; b=8'd6; cin=0; #10;
a=8'd1; b=8'd7; cin=1; #10;
a=8'd5; b=8'd9; cin=1; #10;
$finish;
end
endmodule
