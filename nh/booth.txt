module booth(m,q,product);

    input signed[3:0]m,q;
    output reg signed [7:0]product;
    reg signed [3:0]a,q1;
    reg  q11;
    integer i;
   always @(*)begin
   q11=0;
   a=0;
   q1=q;
   for(i=0;i<4;i=i+1) begin
   if({q1[0],q11}==2'b01)
   a=a+m;
   else if({q1[0],q11}==2'b10)
   a=a-m;
   {a,q1,q11}={a[3],a,q1,q11}>>>1;
   end
   product={a,q1};
   
   
   end
endmodule

module tb;
reg signed [3:0] m,q;
reg signed [7:0] product;

booth uut(.m(m),.q(q),.product(product));
initial begin
$monitor("M=%d | Q=%d | Product =%d ",m,q,product);

m=8'd3;q=8'd4; #10;
m=8'd5;q=8'd1; #10;
m=8'd9;q=8'd2; #10;
$finish;
end
endmodule
