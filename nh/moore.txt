module moor(clk,rst,din,mode,detect);
input clk;
input rst;
input din;
input mode;
output reg detect;
reg[2:0]state,next;
parameter s0=3'b000;
parameter s1=3'b001;
parameter s2=3'b010;
parameter s3=3'b011;
parameter s4=3'b100;

always @(posedge clk)
begin
if(rst)
state<=s0;
else
state<=next;
end
always@(*)
begin
next=state;
case(state)
3'b000: 
if (din==1'b1)
next=s1;
else
next=s0;

3'b001: 
if (din==1'b1)
next=s1;
else
next=s2;

3'b010: 
if (din==1'b1)
next=s3;
else
next=s0;

3'b011: 
if (din==1'b1)
next=s4;
else
next=s2;

3'b100: 

if(mode==1'b1)begin

next=s0;
end
else begin
if (din==1'b1)
next=s1;
else
next=s2;

end
endcase

end
always @(*)
if(state==s4)
detect=1'b1;
else
detect=1'b0;



endmodule




module tb;

wire detect;

$finish;
end
endmodule
