module meal(clk,rst,din,mode,detect);
input clk;
input rst;
input din;
input mode;
output reg detect;
reg[1:0]state,next;
parameter s0=2'b00;
parameter s1=2'b01;
parameter s2=2'b10;
parameter s3=2'b11;


always @(posedge clk)
begin
if(rst)
state<=s0;
else
state<=next;
end
always@(*)
begin
next=state;

detect=0;
case(state)
s0: 
if (din==1'b1)
next=s1;
else
next=s0;

s1: 
if (din==1'b1)
next=s1;
else
next=s2;

s2: 
if (din==1'b1)
next=s3;
else
next=s0;

s3: 
if (din==1'b1)begin
detect=1;
if(mode==0)
  next=s1;
  else
  next=s0;
  end
  else
  next=s2;



endcase

end



endmodule




module tb;
reg din,mode,clk,rst;
wire detect;
meal uut(.din(din),.clk(clk),.mode(mode),.rst(rst),.detect(detect));
always #5 clk=~clk;
initial begin
rst=1;
clk=0;
din=0;mode=0;
$monitor("Reset=%b | mode=%b | din=%b |detect=%b",rst,mode,din,detect);
#10;

din=1; #10;

din=1; #10;
din=1; #10;
din=0; #10;
din=1; #10;
din=1; #10;
#50;

din=1; #10;
din=0; #10;
din=1; #10;
din=1; #10;
din=0; #10;
din=1; #10;
din=1; #10;
$finish;
end
endmodule
