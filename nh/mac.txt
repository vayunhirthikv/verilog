module macunit(clk,rst,a,b,acc);
input clk;
input rst;
input [7:0]a,b;
output reg [15:0]acc;
always@(posedge clk or posedge rst)begin
if(rst)
acc<=4'b0000;
else
acc<=acc+(a*b);
end
endmodule

module tb;
reg clk,rst;
reg [7:0]a,b;
wire[15:0]acc;
macunit uut(.clk(clk),.rst(rst),.a(a),.b(b),.acc(acc));
always #5 clk = ~clk;
initial begin
rst=1;
clk=0;
a=0;b=0;
$monitor(" A=%d | B=%d | Acc =%d ",a,b,acc);
#15;
rst=0;#10;
a=8'd3; b=8'd4; #10;
a=8'd10; b=8'd9; #10;
a=8'd7; b=8'd8; #10;
$finish;
end
endmodule
