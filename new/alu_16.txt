module aluu(a,b,sel,y,carry);

input [7:0] a,b;
input [3:0] sel;
output reg [7:0] y;
output reg carry;

always @(*) begin
    carry=0;
    y=0;

    case(sel)
        4'b0000: y=a;
        4'b0001: {carry,y}=a+1;
        4'b0010: y=a-1;
        4'b0011: {carry,y}=a+b;
        4'b0100: y=a-b;
        4'b0101: y=a*b;
        4'b0110: y=a/b;
        4'b0111: y=a>>1;

        4'b1000: y=~a;
        4'b1001: y=~(a&b);
        4'b1010: y=~(a|b);
        4'b1011: y=a&b;
        4'b1100: y=a|b;
        4'b1101: y=a^b;
        4'b1110: y=~(a^b);
        4'b1111: y=$signed(a)>>>1;

        default: y=0;
    endcase
end

endmodule






module tb;

reg [7:0] a,b;
reg [3:0] sel;
wire [7:0] y;
wire carry;
integer i;

aluu uut(.a(a),.b(b),.sel(sel),.y(y),.carry(carry));

initial begin
    $monitor("A=%d | B=%d | Sel=%b | Y=%d | Carry=%b",
              a,b,sel,y,carry);

    a=8'd20;
    b=8'd5;

    for(i=0;i<16;i=i+1) begin
        sel=i;
        #10;
    end

    $finish;
end

endmodule
