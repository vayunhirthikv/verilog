module booth(
    input  signed [3:0] m, q,
    output reg signed [7:0] product
);
    reg signed [4:0] a;      // 5 bits to avoid overflow
    reg signed [4:0] m5;
    reg        [3:0] q1;
    reg              q0;     // Q(-1)
    integer i;

    always @(*) begin
        a  = 5'sd0;
        m5 = m;              // sign-extends
        q1 = q;
        q0 = 1'b0;
        for (i = 0; i < 4; i = i + 1) begin
            case ({q1[0], q0})
                2'b01:   a = a + m5;
                2'b10:   a = a - m5;
                default: ;
            endcase
            // arithmetic right shift of {a, q1, q0}
            {a, q1, q0} = {a[4], a, q1};
        end
        product = {a[3:0], q1};
    end
endmodule

module booth_tb;

reg signed [3:0] m, q;
wire signed [7:0] product;

booth uut (
    .m(m),
    .q(q),
    .product(product)
);

initial begin
    $monitor("Time=%0t | M=%d | Q=%d | Product=%d",
              $time, m, q, product);

    // Test 1: 3 * 2 = 6
    m = 4'd3;  q = 4'd2;
    #10;

    // Test 2: 4 * (-3) = -12
    m = 4'd4;  q = -4'd3;
    #10;

    // Test 3: (-3) * 2 = -6
    m = -4'd3; q = 4'd2;
    #10;

    // Test 4: (-3) * (-2) = 6
    m = -4'd3; q = -4'd2;
    #10;

    // Test 5: 7 * 3 = 21
    m = 4'd7;  q = 4'd3;
    #10;

    // Test 6: (-8) * 1 = -8
    m = -4'd8; q = 4'd1;
    #10;

    $finish;
end

endmodule
